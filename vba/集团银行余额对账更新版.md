Sub abc()
Dim i, m, g, n, k, k0, k1, c, arr1, arr2, arr4, arr5, brr1, brr2, crr1, crr2
Dim d As Object, d0 As Object, d1 As Object, d2 As Object, wb As Workbook, wba As Workbook
Dim MyPath, MyName
Application.ScreenUpdating = False
Set d = CreateObject("scripting.dictionary")
Set d0 = CreateObject("scripting.dictionary")
Set d1 = CreateObject("scripting.dictionary")
Set d2 = CreateObject("scripting.dictionary")
MyPath = ThisWorkbook.Path
MyName = Dir(MyPath & "\" & "_.xls_")
Do While MyName <> ""
If MyName <> ThisWorkbook.Name Then
Set wb = Workbooks.Open(MyPath & "\" & MyName)
With wb
For g = 1 To .Sheets.Count
If Not IsEmpty(.Sheets(g).UsedRange) Then
d(wb.Name & " " & .Sheets(g).Name) = ""
End If
Next
End With
wb.Close False
End If
MyName = Dir
Loop
k = d.keys 'k:工作簿名称+工作表名称
With ThisWorkbook.Sheets(1)
arr4 = .[a1].Resize(.Range("A1048576").End(xlUp).Row, .Range("XFD1").End(xlToLeft).Column + 1)
End With
For g = 2 To UBound(arr4)
d1(arr4(g, 4) & " " & arr4(g, 5)) = ""
Next
Set wb = Workbooks.Add(xlWBATWorksheet)
With wb.Sheets(1)
.Cells(1, 1) = "工作簿名称"
.Cells(1, 2) = "工作表名称"
.Cells(1, 3) = "差异原因"
g = 2
For i = 0 To d.Count - 1
If Not d0.exists(Split(k(i), " ")(0)) Then
d0(Split(k(i), " ")(0)) = ""
End If
If Not d1.exists(k(i)) Then
.Cells(g, 1) = Split(k(i), " ")(0)
.Cells(g, 2) = Split(k(i), " ")(1)
.Cells(g, 3) = "未找出此银行"
g = g + 1
End If
Next
If .Cells(2, 1) <> "" Then
MsgBox "银行数据汇总异常，需核对"
Exit Sub
Else
wb.Close False
End If
End With
d1.RemoveAll
c = Application.InputBox("请输入对账截止日期", , Format(Date - Day(Date), "YYYY-MM-DD"), , , , , 1)
c = Format(c, "YYYY-MM-DD")
Application.ScreenUpdating = False
k0 = d0.keys
k = 2
Set wba = Workbooks.Add(xlWBATWorksheet)
With wba.Sheets(1)
.Cells(1, 1) = "账簿"
.Cells(1, 2) = "银行账户名称"
.Cells(1, 3) = "用友期末余额"
.Cells(1, 4) = "银行账单期末余额"
.Cells(1, 5) = "差额"
.Cells(1, 6) = "核对结果"
.Cells(1, 7) = "备注"
arr1 = ThisWorkbook.Sheets(2).[a1].CurrentRegion
ReDim arr2(1 To 1, 1 To UBound(arr1, 2))
For i = 2 To UBound(arr1)
For m = 1 To UBound(arr1, 2)
arr2(1, m) = arr1(i, m)
Next
If Not d1.exists(arr1(i, 2) & " " & arr1(i, 5)) Then
d1(arr1(i, 2) & " " & arr1(i, 5)) = arr2
End If
Next
k1 = d1.keys 'k1:账簿+银行账户编码
t1 = d1.Items
For i = 0 To d1.Count - 1
crr1 = Split(k1(i), " ")
If Not d2.exists(crr1(0)) Then
d2(crr1(0)) = Split(crr1(1))
Else
crr2 = d2(crr1(0))
ReDim Preserve crr2(UBound(crr2) + 1)
crr2(UBound(crr2)) = crr1(1)
d2(crr1(0)) = crr2
End If
Next
k2 = d2.keys 'k2:账簿
t2 = d2.Items 't2:银行账户编码
For i = 0 To d2.Count - 1
crr1 = t2(i)
Set wb = Nothing
For m = 2 To UBound(arr4)
If k2(i) = arr4(m, 1) And d0.exists(arr4(m, 4)) Then
Set wb = Workbooks.Open(ThisWorkbook.Path & "\" & arr4(m, 4))
Exit For
End If
Next
If Not wb Is Nothing Then
For g = 0 To UBound(crr1)
a = ""
b = ""
arr5 = d1(k2(i) & " " & crr1(g))
.Cells(k, 1) = arr5(1, 2)
.Cells(k, 2) = arr5(1, 6)
If arr5(1, 19) <> "贷" Then
.Cells(k, 3) = arr5(1, 20)
Else
.Cells(k, 3) = -1 _ arr5(1, 20)
End If
For m = 2 To UBound(arr4)
If k2(i) = arr4(m, 1) And crr1(g) = arr4(m, 2) Then
arr4(m, UBound(arr4, 2)) = "已对账"
If d.exists(arr4(m, 4) & " " & arr4(m, 5)) Then '1、判断是否存在这个工作表
With wb.Sheets(arr4(m, 5))
num1 = .UsedRange.Rows.Count
num2 = .UsedRange.Columns.Count
brr1 = .Cells(1, 1).Resize(num1, num2)
End With
If IsArray(brr1) Then '银行账单有无内容判断
If UBound(brr1) > 1 Then '银行账单有无内容判断
If arr4(m, 8) = "YYYYMMDD" Then
For n = 2 To UBound(brr1)
If Len(brr1(n, arr4(m, 9))) >= 8 Then
brr1(n, arr4(m, 9)) = Left(brr1(n, arr4(m, 9)), 4) & "-" & Mid(brr1(n, arr4(m, 9)), 5, 2) & "-" & Mid(brr1(n, arr4(m, 9)), 7, 2)
End If
Next
Else
For n = 2 To UBound(brr1)
brr1(n, arr4(m, 9)) = Format(brr1(n, arr4(m, 9)), "YYYY-MM-DD")
Next
End If
If arr4(m, 20) = "升序" Then
For n = 1 To UBound(brr1)
If brr1(n, arr4(m, 9)) <= c Then
If brr1(n, arr4(m, 9)) >= a Then
a = brr1(n, arr4(m, 9))
b = Val(Replace(brr1(n, arr4(m, 12)), ",", ""))
End If
End If
Next
Else
For n = UBound(brr1) To 1 Step -1
If brr1(n, arr4(m, 9)) <= c Then
If brr1(n, arr4(m, 9)) >= a Then
a = brr1(n, arr4(m, 9))
b = Val(Replace(brr1(n, arr4(m, 12)), ",", ""))
End If
End If
Next
End If
End If
End If
If b <> "" Then
.Cells(k, 4) = Round(b, 2)
End If
Else
.Cells(k, 7) = "没有该银行账单"
End If '1、判断结束
k = k + 1
Exit For
Else
If m = UBound(arr4) Then
.Cells(k, 7) = "没有该银行账单"
k = k + 1
End If
End If
Next
Next
wb.Close False
Else
For g = 0 To UBound(crr1)
arr5 = d1(k2(i) & " " & crr1(g))
.Cells(k, 1) = arr5(1, 2)
.Cells(k, 2) = arr5(1, 6)
If arr5(1, 19) <> "贷" Then
.Cells(k, 3) = arr5(1, 20)
Else
.Cells(k, 3) = -1 _ arr5(1, 20)
End If
.Cells(k, 7) = "没有该银行账单"
k = k + 1
Next
End If
Next
For m = 2 To UBound(arr4)
a = ""
b = ""
If arr4(m, UBound(arr4, 2)) = "" And d.exists(arr4(m, 4) & " " & arr4(m, 5)) Then
Set wb = Workbooks.Open(ThisWorkbook.Path & "\" & arr4(m, 4))
With wb.Sheets(arr4(m, 5))
num1 = .UsedRange.Rows.Count
num2 = .UsedRange.Columns.Count
brr1 = .Cells(1, 1).Resize(num1, num2)
End With
If IsArray(brr1) Then '银行账单有无内容判断
If UBound(brr1) > 1 Then '银行账单有无内容判断
If arr4(m, 8) = "YYYYMMDD" Then
For n = 2 To UBound(brr1)
If Len(brr1(n, arr4(m, 9))) >= 8 Then
brr1(n, arr4(m, 9)) = Left(brr1(n, arr4(m, 9)), 4) & "-" & Mid(brr1(n, arr4(m, 9)), 5, 2) & "-" & Mid(brr1(n, arr4(m, 9)), 7, 2)
End If
Next
Else
For n = 2 To UBound(brr1)
brr1(n, arr4(m, 9)) = Format(brr1(n, arr4(m, 9)), "YYYY-MM-DD")
Next
End If
If arr4(m, 20) = "升序" Then
For n = 1 To UBound(brr1)
If brr1(n, arr4(m, 9)) <= c Then
If brr1(n, arr4(m, 9)) >= a Then
a = brr1(n, arr4(m, 9))
b = Val(Replace(brr1(n, arr4(m, 12)), ",", ""))
End If
End If
Next
Else
For n = UBound(brr1) To 1 Step -1
If brr1(n, arr4(m, 9)) <= c Then
If brr1(n, arr4(m, 9)) >= a Then
a = brr1(n, arr4(m, 9))
b = Val(Replace(brr1(n, arr4(m, 12)), ",", ""))
End If
End If
Next
End If
End If
End If
If b <> "" Then
.Cells(k, 1) = arr4(m, 1)
.Cells(k, 2) = arr4(m, 3)
.Cells(k, 4) = Round(b, 2)
k = k + 1
End If
wb.Close False
End If
Next
For m = 2 To .Cells(.Rows.Count, 1).End(xlUp).Row
If .Cells(m, 3) = "" Then
.Cells(m, 3) = 0
End If
If .Cells(m, 4) = "" Then
.Cells(m, 4) = 0
End If
.Cells(m, 5) = Round(.Cells(m, 3) - .Cells(m, 4), 2)
If .Cells(m, 5) = 0 Then
.Cells(m, 6) = "无误"
Else
.Cells(m, 6) = "异常"
End If
Next
End With
Application.ScreenUpdating = True
MsgBox "已完成"
End Sub
